    <script>
        // Array of all sub-question textarea IDs
        const ALL_NOTE_IDS = [
            'edu-sub-1a', 'edu-sub-1b', 'edu-sub-2a', 'edu-sub-2b', 'edu-sub-3a', 'edu-sub-3b',
            'env-sub-1a', 'env-sub-1b', 'env-sub-2a', 'env-sub-2b', 'env-sub-3a', 'env-sub-3b',
            'hob-sub-1a', 'hob-sub-1b', 'hob-sub-2a', 'hob-sub-2b', 'hob-sub-3a', 'hob-sub-3b'
        ];

        let currentLang = 'en'; // Default language

        // Toggle Language Function
        function toggleLanguage() {
            if (currentLang === 'en') {
                currentLang = 'ja';
                document.querySelectorAll('.lang-en').forEach(el => el.classList.add('hidden'));
                document.querySelectorAll('.lang-ja').forEach(el => el.classList.remove('hidden'));
                document.getElementById('lang-btn-text').textContent = 'English に切り替え';
            } else {
                currentLang = 'en';
                document.querySelectorAll('.lang-ja').forEach(el => el.classList.add('hidden'));
                document.querySelectorAll('.lang-en').forEach(el => el.classList.remove('hidden'));
                document.getElementById('lang-btn-text').textContent = '日本語に切り替え';
            }
        }

        // Switch Navigation Tabs
        function switchTab(tabId) {
            document.querySelectorAll('main > section').forEach(sec => sec.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
            
            document.getElementById('content-' + tabId).classList.remove('hidden');
            document.getElementById('tab-' + tabId).classList.add('active');
        }

        // Save single note to localStorage
        function saveNote(id) {
            const el = document.getElementById(id);
            if (el) {
                localStorage.setItem('rv_prep_note_' + id, el.value);
                showStatus(currentLang === 'en' ? 'Saved to local device memory!' : '端末にメモが自動保存されました！');
            }
        }

        // Save all notes explicitly
        function saveAllNotes() {
            ALL_NOTE_IDS.forEach(id => {
                const el = document.getElementById(id);
                if (el) {
                    localStorage.setItem('rv_prep_note_' + id, el.value);
                }
            });
            showStatus(currentLang === 'en' ? 'All notes successfully saved!' : 'すべてのメモが保存されました！');
        }

        // Load all saved notes on startup
        function loadAllNotes() {
            ALL_NOTE_IDS.forEach(id => {
                const savedVal = localStorage.getItem('rv_prep_note_' + id);
                const el = document.getElementById(id);
                if (savedVal && el) {
                    el.value = savedVal;
                }
            });
        }

        // Clear all notes
        function clearAllNotes() {
            const confirmMsg = currentLang === 'en' 
                ? "Are you sure you want to delete all your prepared notes?" 
                : "本当にすべてのメモを消去しますか？";
            
            if (confirm(confirmMsg)) {
                ALL_NOTE_IDS.forEach(id => {
                    localStorage.removeItem('rv_prep_note_' + id);
                    const el = document.getElementById(id);
                    if (el) el.value = '';
                });
                showStatus(currentLang === 'en' ? 'All notes deleted.' : 'すべてのメモが消去されました。');
            }
        }

        // Status banner helper
        let statusTimeout;
        function showStatus(msg) {
            const banner = document.getElementById('status-message');
            const textEl = document.getElementById('status-text');
            textEl.textContent = msg;
            banner.classList.remove('hidden');

            clearTimeout(statusTimeout);
            statusTimeout = setTimeout(() => {
                banner.classList.add('hidden');
            }, 2500);
        }

        function dismissStatus() {
            document.getElementById('status-message').classList.add('hidden');
        }

        // Page load initialization
        window.onload = function() {
            loadAllNotes();
        };
    </script>
